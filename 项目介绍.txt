from bs4 import BeautifulSoup

def extract_douyin_info(url):
    # 解析分享链接获取视频信息
    headers = {'User-Agent': 'Mozilla/5.0...'}
    resp = requests.get(url, headers=headers, allow_redirects=True)

import sqlite3
from datetime import datetime

class KnowledgeBase:
    def __init__(self):
        self.conn = sqlite3.connect('douyin_kb.db')
        self._init_db()
    
    def add_video(self, video_data):
        # 存储到SQLite，支持全文搜索
        cursor = self.conn.cursor()
        cursor.execute('''
            INSERT INTO videos 
            (url, title, summary, category, tags, added_at)
            VALUES (?, ?, ?, ?, ?, ?)
        ''', (
            video_data['url'],
            video_data['title'],
            video_data['summary'],
            video_data['category'],
            ','.join(video_data.get('tags', [])),
            datetime.now()
        ))
        self.conn.commit()
    
    def search(self, query):
        # 支持语义搜索（需配合embedding模型）
        cursor = self.conn.cursor()
        cursor.execute('''
            SELECT * FROM videos 
            WHERE summary MATCH ? OR title MATCH ?
        ''', (query, query))
        return cursor.fetchall()

AI 引擎封装（支持多模型切换）：
# backend/app/ai/summarizer.py
import openai
import json
from typing import Optional

class AIProcessor:
    def __init__(self, provider: str = "ollama"):
        self.provider = provider
        # Ollama 本地服务（默认端口11434）
        if provider == "ollama":
            self.client = openai.OpenAI(base_url="http://localhost:11434/v1", api_key="ollama")
            self.model = "qwen2.5:14b"  # 中文表现好，14B 在 16G 显存可跑
        else:
            self.client = openai.OpenAI(api_key="your-api-key")
            self.model = "gpt-4o-mini"
    
    async def process_video(self, title: str, desc: str, author: str) -> dict:
        prompt = SUMMARY_PROMPT.format(title=title, desc=desc, author=author)
        
        resp = self.client.chat.completions.create(
            model=self.model,
            messages=[{"role": "user", "content": prompt}],
            temperature=0.3,
            response_format={"type": "json_object"} if self.provider != "ollama" else None
        )
        
        return json.loads(resp.choices[0].message.content)

项目简介
这是一个专为重度抖音用户设计的本地私有化知识管理工具。它通过浏览器自动化技术抓取你的抖音收藏视频，利用 AI 对视频内容进行智能总结、自动分类和标签提取，最终构建成一个支持语义搜索的结构化知识库。所有数据存储在本地，无需担心隐私泄露，也无需依赖抖音官方 API。
核心能力
能力	说明
一键同步	自动登录抖音网页版，抓取收藏列表，支持断点续传和增量更新
AI 智能总结	基于大语言模型（支持本地 Ollama / 云端 API）自动生成视频核心要点、干货评分
自动分类体系	将杂乱的收藏自动归类到「编程技术 / 商业思维 / 生活技巧 / 心理学」等知识维度
语义搜索引擎	不是简单的关键词匹配——输入”如何提高专注力”，能搜到相关的心理学和学习方法视频
私有化部署	前后端完全本地运行，视频数据、AI 处理、向量数据库均不出境
现代化看板	React 驱动的可视化界面，支持分类浏览、时间轴、标签云和质量筛选
技术架构
后端：Python FastAPI + Playwright（浏览器自动化）+ SQLAlchemy（SQLite）+ ChromaDB（向量检索）
前端：React 18 + TypeScript + Tailwind CSS + Shadcn/ui
AI 层：支持 Ollama 本地模型（推荐 Qwen2.5 / DeepSeek）或 OpenAI / DeepSeek API
数据流：Playwright 抓取 → AI 总结与向量化 → SQLite 结构化存储 + ChromaDB 语义索引 → React 可视化消费
适用场景
•信息囤积症自救：收藏了 500+ 视频却从未回看，现在可以用 3 秒找到半年前收藏的某个技术教程
•跨领域学习整理：同时关注编程、商业、健身、心理学，需要按主题建立个人知识图谱
•内容创作者选题库：将收藏的爆款视频结构化，分析其标签和核心卖点，辅助自己的内容策划
•本地知识备份：防止抖音账号异常或视频下架导致收藏丢失，建立本地可导出的知识档案
设计理念
「收藏不等于拥有，检索才能创造价值。」
我们不追求替代抖音的娱乐体验，而是解决「收藏后遗忘」的信息管理痛点。通过 AI 将非结构化的短视频内容转化为结构化的知识节点，让你的每一次收藏都真正沉淀为认知资产。
